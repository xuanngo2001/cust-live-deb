# List and sort directories by size.
    du -c --max-depth=1 | sort -k1,1n | numfmt --field=1 --from-unit=1024 --to=iec-i  --format='%10f' --suffix B

# List size of directories (largest first)
    du -sh -- * | sort -rh

# List directories size, excluding system directories, virtual file systems, and mounted external drives from your disk space check, use the -x (one-file-system) flag
#   /proc, /sys, /dev, /run
    sudo du -h -x --max-depth=1 /var/lib/docker/containers | sort -rh
    sudo find /var/lib/docker/containers/ -type f -name "*-json.log" -exec ls -lh {} +
    sudo find /var/lib/docker/containers/ -type f -name "*-json.log" -ok rm {} \;
