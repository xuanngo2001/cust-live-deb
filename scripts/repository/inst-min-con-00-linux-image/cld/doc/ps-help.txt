# Find application name
    ps aux | grep APP_NAME
    
# Display PID
    ps -eo pid,command | grep APP_NAME
    
# Which process is using the most RAM
    ps -e -o pid,vsz,comm= | sort -n -k 2
    ps aux  | awk '{print $6/1024 " MB\t\t" $11}'  | sort -n


# Kill processes
    # killall requires an exact match of the process name. It matches the first 15 characters.
    killall apache2

    # pkill uses pattern matching (regular expressions) against various process attributes, 
    #   including the process name, command line, PID, PPID, and UID. This allows for more flexible and partial matching.
    pkill apache2
